### Полное программное решение. Случай, где  все значения переменных в таблице известны

```python
from itertools import *

def f(x, y, z): # Вводим актуальные переменные
	return (x <= y) and (y <= z) # Вводим актуальное выражение

table = [(1, 0, 0),(1, 0, 1)] # Вводим актуальную таблицу по условию, по порядку

for p in permutations('xyz'): # Для каждого варианта перестановки мы проверяем соответствующие значения
	if [f(**dict(zip(p, row))) for row in table] == [0, 1]: # [0, 1] это значения F
		print(''.join(p)) # Можно без join
```

### Полное программное решение. Случай, где  не все значения переменных в таблице известны

```python
from itertools import *
  
def f(x, y, z, w):
	return ((y <= x) or (not z and w)) == (w == x)

for a in product([0, 1], repeat=3): # repeat = кол-во пустот
	table = [(a[0], 1, 0, 0), (0, 0, 0, 1), (0, 1, a[1], a[2])]
	if len(table) == len(set(table)): # Исключение идентичных строк
		for p in permutations('xywz'):
			if [f(**dict(zip(p, r))) for r in table] == [1, 1, 1]:
				print(''.join(p))
```

### Полное программное решение. Случай, где  F1 и F2

```python
from itertools import *

def f1(x, y, w, z):
	return (w or not y) <= (z == x)
  
def f2(x, y, w, z):
	return (w or not y) == (x <= z)
  
for a in product([0, 1], repeat=4): # repeat = кол-во пустот
	table = [(a[0], 1, a[1], 1), (a[2], 0, 0, 0), (0, 0, 0, a[3])]
	if len(table) == len(set(table)): # Исключение идентичных строк
		for p in permutations('xywz'):
			for b in product([0, 1], repeat=2):
				if [f1(**dict(zip(p, r))) for r in table] == [0, b[0], 0]:
					if [f2(**dict(zip(p, r))) for r in table] == [b[1], 0, 0]:
						print(''.join(p))
```
