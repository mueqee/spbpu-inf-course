a = [int(x) for x in open('17.txt')]

min_d = min([x for x in a if 9 < x < 100]) # Двузначное число

c = []

for i in range(len(a) - 1):
	if ((9 < a[i] < 100) + (9 + a[i + 1] < 100)) == 1:
		if (a[i] + a[i + 1]) % min_d == 0:
			c.append(a[i] +[i + 1])

print(len(c), max(c))
