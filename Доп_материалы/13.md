```python
from ipaddress import ip_network

# Адрес узла (компьютер), сети, маски

net = ip_network('218.194.82.148/255.255.255.192', 0)

for ip in net.hosts(): # Пройтись по всем адресам этой сети

print(ip) # Выведутся все доступные адреса данной сети

# Если просто пройтись по net без указания hosts, то необходимо
# обязательно исключить начальный и конечный адреса
```

Если в условии говорится про все адреса сети, то перебираем все в net, но если в условии говорится об адресах конкретных устройств, то мы перебираем в net.hosts()

```python
from ipaddress import ip_network

net = ip_network('5.2.5.0/255.255.0.0', 0)
k = 0

for ip in net.hosts():
	ip_binary = bin(int(ip))[2:].zfill(32)
	if ip_binary.count('0') % 25 == 0:
		k += 1

print(k)
```

from ipaddress import ip_network

# В случае, если по условию не задана маска, то ее можно
# представить в виде чисел от 0 до 32

# Если дан адрес узла и сети, то берем сначала адрес узла

for m in range(0, 33):
	net = ip_network(f'218.48.192.56/{m}', 0)
	if str(net.network_address) == '218.48.192.0':
		if len(list(net.hosts())) >= 500:
			print(net.netmask)
