
Для каждого файла создаем диаграмму (точки). Дополнительно для удобства указываем все две оси. В случае, если шаг слишком большой, самостоятельно задаем (1 или 2). После создания диаграмм обратно переходим в текстовые файлы и заменяем все запятые на точки.

Под абсолютной величиной имеется ввиду число под модулем (без минуса)

```python
from math import dist

# Находим центр
def centre(cl):
	m = []
	for p in cl:
		# dist считает расстояние между точками. Формула дано по условию, но в питоне есть сразу готовая функция
		s = sum(dist(p, p1) for p1 in cl) # Считаем сумму растояний между точками
		m.append([s, p]) # Добавляе в список расстояние и сами точки
	return min(m)[1]

clA = [[], []] # По условию в файле А хранятся точки ДВУХ кластерах

for s in open('27A.txt'):
	x, y = [float(d) for d in s.split()]
	if x > 20 or y > 25: # Отбрасываем аномальные точки
		pass
	elif y > 15:
		clA[0].append([x, y])
	else:
		clA[1].append([x, y])

clB = [[], [], []] # По условию в файле В точки хранятся в трех кластерах
for s in open('27B.txt'):
	x, y = [float(d) for d in s.split()]
	if (x > -3.5 and x < -2) or (x > 0.5):
		pass
	elif x > -2 and x < -1:
		clB[0].append([x, y])
	elif y >= 5 and y < 6:
		clB[1].append([x, y])
	else:
		clB[2].append([x, y])

# По условию следующее: для файла А определить координаты центра каждого кластеров
# После найти два числа: Рх минимальную из абсцисс центров кластеров и Ру -
# минимальную из ординат центров кластеров
# Для файла В найти Q1 - минимальное количество точек в кластере и Q2 - максимальное
  
x0, y0 = centre(clA[0])
x1, y1 = centre(clA[1])

p = [len(cl) for cl in clB]

# В ответ записать следующее: Рх * 10000, Ру * 10000
print(int(min(x0, x1) * 10000), int(min(y0, y1) * 10000))
print(min(p), max(p))
```

```python
from math import dist

def centre(cl):
	m = []
	for p in cl:
		s = sum(dist(p, p1) for p1 in cl)
		m.append([s, p])
	return min(m)[1]

clA = [[], []]
for s in open('27A.txt'):
	x, y = [float(d) for d in s.split()]
	if x < -5 or y < -10:
		pass
	elif x < 0:
		clA[0].append([x, y])
	else:
		clA[1].append([x, y])

clB = [[], [], []]
for s in open('27B.txt'):
	x, y = [float(d) for d in s.split()]
	if x < -2 or x > 0:
		pass
	elif x < -1:
		clB[0].append([x, y])
	elif y > 5:
		clB[1].append([x, y])
	else:
		clB[2].append([x, y])

x0, y0 = centre(clA[0])
x1, y1 = centre(clA[1])

x2, y2 = centre(clB[0])
x3, y3 = centre(clB[1])
x4, y4 = centre(clB[2]) 

qx = (x2 + x3 + x4)/3
qy = (y2 + y3 + y4)/3

print(int(max(x0, x1) * 10000), int(max(y0, y1) * 10000))
print(int(qx * 10000), int(qy * 10000))
```
